class DailyForecast {
  const DailyForecast({
    required this.day,
    required this.low,
    required this.high,
    required this.condition,
  });

  final DateTime day;
  final Temperature low;
  final Temperature high;
  final WeatherCondition condition;

  double get spread => high.celsius - low.celsius;
}

class Forecast {
  const Forecast(this.days);

  final List<DailyForecast> days;

  DailyForecast? get warmest => days.isEmpty
      ? null
      : days.reduce((a, b) => a.high.celsius >= b.high.celsius ? a : b);

  double get averageSpread => days.isEmpty
      ? 0
      : days.map((d) => d.spread).reduce((a, b) => a + b) / days.length;
}

import 'dart:convert';

import 'package:http/http.dart' as http;
import 'package:weather_sample/weather/data/api_exceptions.dart';
import 'package:weather_sample/weather/data/weather_response.dart';
import 'package:weather_sample/weather/models/location.dart';

class WeatherApiClient {
  WeatherApiClient({http.Client? httpClient})
      : _httpClient = httpClient ?? http.Client();

  static const _geocodingHost = 'geocoding-api.open-meteo.com';
  static const _weatherHost = 'api.open-meteo.com';

  final http.Client _httpClient;

  Future<Location> locationSearch(String query) async {
    final json = await _get(_geocodingHost, '/v1/search', {'name': query});
    final results = json['results'] as List?;
    if (results == null || results.isEmpty) throw LocationNotFoundFailure();
    return Location.fromJson(results.first as Map<String, dynamic>);
  }

  Future<WeatherResponse> getWeather(Location location) async {
    final json = await _get(_weatherHost, '/v1/forecast', {
      'latitude': '${location.latitude}',
      'longitude': '${location.longitude}',
      'current_weather': 'true',
    });
    final current = json['current_weather'] as Map<String, dynamic>?;
    if (current == null) throw WeatherNotFoundFailure();
    return WeatherResponse.fromJson(current);
  }

  Future<Map<String, dynamic>> _get(
    String host,
    String path,
    Map<String, String> query,
  ) async {
    final response = await _httpClient.get(Uri.https(host, path, query));
    if (response.statusCode != 200) throw RequestFailure(response.statusCode);
    return jsonDecode(response.body) as Map<String, dynamic>;
  }

  void close() => _httpClient.close();
}

        >= 71 && <= 77 || 85 || 86 => WeatherCondition.snowy,

import 'package:weather_sample/weather/models/weather.dart';

mixin Expiring {
  Duration get maxAge => const Duration(minutes: 10);

  bool isFresh(DateTime time, DateTime now) => now.difference(time) < maxAge;
}

class WeatherCache with Expiring {
  final _entries = <String, Weather>{};

  Weather? read(String city, DateTime now) {
    final weather = _entries[city.toLowerCase()];
    if (weather == null || !isFresh(weather.lastUpdated, now)) return null;
    return weather;
  }

  void write(String city, Weather weather) =>
      _entries[city.toLowerCase()] = weather;

  void clear() => _entries.clear();
}

import 'package:weather_sample/weather/data/weather_api_client.dart';
import 'package:weather_sample/weather/data/weather_cache.dart';
import 'package:weather_sample/weather/models/temperature.dart';
import 'package:weather_sample/weather/models/weather.dart';

class WeatherRepository {
  WeatherRepository({required WeatherApiClient apiClient, WeatherCache? cache})
      : _apiClient = apiClient,
        _cache = cache ?? WeatherCache();

  final WeatherApiClient _apiClient;
  final WeatherCache _cache;

  Future<Weather> getWeather(String city) async {
    final now = DateTime.now();
    final cached = _cache.read(city, now);
    if (cached != null) return cached;
    final location = await _apiClient.locationSearch(city);
    final response = await _apiClient.getWeather(location);
    final weather = Weather(
      condition: response.condition,
      lastUpdated: now,
      location: location.name,
      temperature: Temperature(celsius: response.temperature),
    );
    _cache.write(city, weather);
    return weather;
  }

  void dispose() {
    _cache.clear();
    _apiClient.close();
  }
}

import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:weather_sample/settings/temperature_units.dart';
import 'package:weather_sample/weather/cubit/weather_state.dart';
import 'package:weather_sample/weather/data/weather_repository.dart';

class WeatherCubit extends Cubit<WeatherState> {
  WeatherCubit(this._repository) : super(WeatherState());

  final WeatherRepository _repository;

  Future<void> fetchWeather(String? city) async {
    if (city == null || city.isEmpty) return;
    emit(state.copyWith(status: WeatherStatus.loading));
    try {
      final weather = await _repository.getWeather(city);
      emit(state.copyWith(status: WeatherStatus.success, weather: weather));
    } on Exception {
      emit(state.copyWith(status: WeatherStatus.failure));
    }
  }

  Future<void> refreshWeather() async {
    if (state.status != WeatherStatus.success) return;
    await fetchWeather(state.weather.location);
  }

  void changeUnits(TemperatureUnits units) => emit(state.copyWith(units: units));
}

class _SearchPageState extends State<SearchPage> {
  final _controller = TextEditingController();
  String? _error;

  void _submit() {
    final error = CityValidator.validate(_controller.text);
    if (error != null) return setState(() => _error = error);
    Navigator.of(context).pop(CityValidator.normalize(_controller.text));
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('City search')),
      body: TextField(
        controller: _controller,
        decoration: InputDecoration(errorText: _error),
        onSubmitted: (_) => _submit(),
      ),
    );
  }
